// Function to get the last two letters of a word
function getLastTwoLetters(word) {
    if (word.length < 2) {
        return word;
    } else {
        return word.substring(word.length - 2);
    }
}

// Function to find rhymes for a given word
function suggestRhymes(wordList, word) {
    const Inputword = getLastTwoLetters(word);
    const rhymes = [];

    for (let i = 0; i < wordList.length; i++) {
        const Listword = wordList[i];
        if (Listword === word) {
            continue;
        }

        const Listwordcheck = getLastTwoLetters(Listword);
        if (Listwordcheck === Inputword) {
            rhymes.push(Listword);
        }
    }

    if (rhymes.length === 0) {
        return "No rhymes found.";
    } else {
        return rhymes;
    }
}
